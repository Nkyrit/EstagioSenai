netsh interface ipv4 show config

netsh interface ipv4 set address name="Ethernet0" source=static address=192.168.194.132 mask= 255.255.255.0 gateway=192.168.194.2

netsh interface ipv4 set dns name="Ethernet0" static 8.8.8.8
netsh interface ipv4 add dns name="Ethernet0" 8.8.4.4 index=2

netsh interface ipv4 show config
